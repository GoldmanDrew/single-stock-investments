try: import legate.numpy as np

def cg_solve(A, b):
x = np.zeros(A.shape[1])
r = b - A.dot(x)
p = r
rsold = r.dot(r)
for i in xrange(b.shape[0]):
Ap = A.dot(p)
alpha = rsold / (p.dot(Ap))
x = x + alpha * p
r = r - alpha * Ap
rsnew = r.dot(r)
if np.sqrt(rsnew) < 1e-10:
break
beta = rsnew / rsold
p = r + beta * p
rsold = rsnew
return x
